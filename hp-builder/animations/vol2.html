const CH = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ#$%&@01';
function scramble(el, text){
  let f = 0;
  const start = [...text].map(() => Math.floor(Math.random()*18));  // 文字ごとの確定タイミング
  (function tick(){
    el.innerHTML = [...text].map((c,i) =>
      f > start[i] + 12 ? c
      : f > start[i]    ? `<span class="r">${CH[Math.floor(Math.random()*CH.length)]}</span>`
      : ' ').join('');
    if(f++ < 46) requestAnimationFrame(tick);
    else el.textContent = text;
  })();
}

// JS：後ろに回ったカードを縮めて暗くする
// ※ sticky で止まった要素は top が動かなくなるので、自分の位置では進捗を測れない。
//    「次のカードがどれだけ覆ってきたか」で測るのが正解。
cards.forEach((c, i) => {
  const next = cards[i+1];
  if(!next) return;
  const passed = clamp(1 - (next.getBoundingClientRect().top
                          - c.getBoundingClientRect().top) / c.offsetHeight);
  c.style.transform = `scale(${1 - passed * 0.12})`;
  c.style.filter    = `brightness(${1 - passed * 0.45})`;
});

18 / TRANSITION

ページが切り替わる瞬間を演出する

押してみてください。5枚のパネルが時間差で降りて、画面を覆ってから開きます。 実際のサイトでは、覆っている間に次ページを読み込みます。

SPAでなくても、<a> のクリックを止めてアニメ後に location.href を書き換えれば同じ体験になります。 戻るボタンで戻ったときにオーバーレイが残らないよう、pageshow で必ず解除しておくのが実務のコツです。
コード 
link.addEventListener('click', e => {
  e.preventDefault();
  curtain.classList.add('go');                       // パネルが降りる
  setTimeout(() => location.href = link.href, 620);  // 覆いきってから遷移
});
addEventListener('pageshow', () => curtain.classList.remove('go'));  // 戻る対策

#curtain div{transform:scaleY(0);transform-origin:50% 100%;
  transition:transform .55s cubic-bezier(.76,0,.24,1)}
#curtain.go div{transform:scaleY(1);transform-origin:50% 0%}
